namespace findaname
{
    public partial class Form1 : Form
    {
        public Form1()
        {
            InitializeComponent();
        }

        private void lblentername_Click(object sender, EventArgs e)
        {
        }

        private void txtname_TextChanged(object sender, EventArgs e)
        {
        }

        private void btnsearch_Click(object sender, EventArgs e)
        {
            string[] names = { "Ana", "Ben", "Carlo", "David" };

            string searchName = txtname.Text;
            bool found = false;

            foreach (string name in names)
            {
                if (name.Equals(searchName, StringComparison.OrdinalIgnoreCase))
                {
                    Console.WriteLine("Found!");
                    lblresult.Text = "Found!";
                    found = true;
                }
            }

            if (!found)
            {
                lblresult.Text = "Name not found.";
            }
        }

        private void lblresult_Click(object sender, EventArgs e)
        {
        }
    }
}

BINARY SEARCH 



namespace binarysearch
{
    public partial class Form1 : Form
    {
        public Form1()
        {
            InitializeComponent();

            int[] numbers = { 10, 20, 30, 40, 50, 60, 70 };
            int left = 0;
            int right = numbers.Length - 1;

            while (left <= right)
            {
                int middle = (left + right) / 2;

                if (numbers[middle] == 50)
                {
                    lblresult.Text = "Found!";
                    break;
                }
                else if (numbers[middle] < 50)
                {
                    left = middle + 1;
                }
                else
                {
                    right = middle - 1;
                }
            }
        }

        private void lblresult_Click(object sender, EventArgs e)
        {
        }
    }
}
